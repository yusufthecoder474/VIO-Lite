cmake_minimum_required(VERSION 3.18)
project(VIO_Lite VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VIO_OPENCV_PREFIX "" CACHE PATH "Optional extracted OpenCV installation prefix")
if(VIO_OPENCV_PREFIX)
 find_path(OpenCV_INCLUDE_DIRS opencv2/core.hpp PATHS "${VIO_OPENCV_PREFIX}/include/opencv4" NO_DEFAULT_PATH REQUIRED)
 foreach(component core imgproc imgcodecs video calib3d features2d flann)
  find_library(OPENCV_${component} opencv_${component} PATHS "${VIO_OPENCV_PREFIX}/lib/x86_64-linux-gnu" "${VIO_OPENCV_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  list(APPEND OpenCV_LIBS "${OPENCV_${component}}")
 endforeach()
else()
 find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs video calib3d features2d)
endif()
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(vio src/dataset.cpp src/imu.cpp src/vision.cpp src/scale.cpp src/output.cpp)
target_include_directories(vio PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vio PUBLIC Eigen3::Eigen ${OpenCV_LIBS})
if(MSVC)
 target_compile_options(vio PRIVATE /W4)
else()
 target_compile_options(vio PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(VIO_SANITIZE "Enable ASan and UBSan on GCC/Clang" OFF)
if(VIO_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(vio PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(vio PUBLIC -fsanitize=address,undefined)
endif()
add_executable(vio_lite src/main.cpp)
target_link_libraries(vio_lite PRIVATE vio)
include(CTest)
if(BUILD_TESTING)
 add_executable(vio_tests tests/test_core.cpp)
 target_link_libraries(vio_tests PRIVATE vio)
 foreach(case parsing ordering interpolation invalid calibration tracking composition scale degenerate csv imu initialization)
  add_test(NAME ${case} COMMAND vio_tests ${case})
 endforeach()
endif()
